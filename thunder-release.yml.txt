name: Thunder Release

# THUNDER: personal build release for BRAT installs.
# BRAT installs from GitHub releases, not branches, so this builds,
# tests, and publishes a non-draft release carrying main.js,
# manifest.json, and styles.css. Tag format: 0.5.25-thunder.<n>
# (upstream version 0.5.25 plus the Thunder build number).

on:
  push:
    branches: [thunder]
  workflow_dispatch:
    inputs:
      release_tag:
        description: "Release tag (blank = no release, build and test only)"
        required: false
        default: ""

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Use Node.js 20
        uses: actions/setup-node@v4
        with:
          node-version: 20

      - name: Install
        run: npm install

      - name: Test
        run: npm test

      - name: Build
        run: npm run build

      - name: Verify PRO features are in the built main.js
        run: |
          for term in googledrive onedrivefull azureblobstorage smart_conflict box pcloud yandex koofr; do
            grep -q "$term" main.js && echo "found: $term" || (echo "MISSING: $term" && exit 1)
          done

      - name: Set release version in manifest and package
        if: github.event_name == 'workflow_dispatch' && inputs.release_tag != ''
        run: |
          TAG="${{ inputs.release_tag }}"
          VERSION="${TAG#v}"
          node -e "const fs=require('fs');const m=JSON.parse(fs.readFileSync('manifest.json'));m.version='$VERSION';fs.writeFileSync('manifest.json',JSON.stringify(m,null,2)+'\n');const p=JSON.parse(fs.readFileSync('package.json'));p.version='$VERSION';fs.writeFileSync('package.json',JSON.stringify(p,null,2)+'\n');"
          cat manifest.json

      - name: Package zip
        if: github.event_name == 'workflow_dispatch' && inputs.release_tag != ''
        run: zip thunder-sync.zip main.js manifest.json styles.css

      - name: Publish non-draft release for BRAT
        if: github.event_name == 'workflow_dispatch' && inputs.release_tag != ''
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          gh release create "${{ inputs.release_tag }}" main.js manifest.json styles.css thunder-sync.zip \
            --title "${{ inputs.release_tag }}" \
            --notes "Thunder personal build of Remotely Save (upstream 0.5.25). All features kept, PRO subscription check neutralized locally. See THUNDER.md. Install with BRAT: thunder-mythos/remotely-save. Do not update this plugin from the community plugin browser afterwards, or upstream replaces this build." \
            --latest
